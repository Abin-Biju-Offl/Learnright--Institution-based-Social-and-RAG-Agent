# code-data-boundaries — templates

## Deploy exclude list (rsync), sourced by every sync script

```bash
# deploy/excludes.sh — the single source of truth for what "code" is.
# Keep the DATA / SECRET / RUNTIME block in lockstep with .gitignore (tests/test_lockstep.py).
# The WORKBENCH block deliberately does NOT mirror .gitignore: those files are tracked on
# purpose and simply have no audience in production.
SHIP_EXCLUDES=(
  --exclude=.git                 # no trailing slash: in a worktree .git is a FILE
  # --- WORKBENCH (anchored with a leading slash, or it matches at any depth) ---
  --exclude=/.claude/
  --exclude=/.github/
  --exclude=/CLAUDE.md
  --exclude=/AGENTS.md
  --exclude=/README.md
  --exclude=/SPEC.md
  --exclude=/design.md
  --exclude=/LOG.md
  --exclude=/docs/
  # --- DATA (derived; regenerated by adapters) ---
  --exclude='tenants/*/canonical/'
  --exclude='**/.*.tmp-*'        # in-flight atomic writes
  --exclude='**/run_outputs/'
  # --- SECRETS ---
  --exclude=.env
  --exclude=deploy/env/          # per-tenant env sources: every tenant's tokens
  # --- RUNTIME STATE ---
  --exclude=.venv                # no trailing slash: may be a symlink in a worktree
  --exclude=__pycache__/
  --exclude=var/                 # locks, sessions, logs, outputs
  # --- SERVICE USER HOME (if $HOME is the repo root) ---
  --exclude=.config/
  --exclude=.cache/
  --exclude=.local/
  --exclude=.ssh/
)
```

## Lockstep test (set equality, not spot checks)

```python
import re
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]

# Written out literally, so a pattern deleted from BOTH lists is still caught here.
DATA_AND_SECRETS = {"tenants/*/canonical/", "**/.*.tmp-*", "**/run_outputs/", ".env", "deploy/env/"}

gitignore = {l.strip() for l in (ROOT / ".gitignore").read_text().splitlines()
             if l.strip() and not l.startswith("#")}
sync = (ROOT / "deploy/excludes.sh").read_text()
block = sync.split("# --- DATA", 1)[1].split("# --- RUNTIME STATE", 1)[0]   # data + secrets
excluded = {m.strip("'\"") for m in re.findall(r"--exclude=(\S+)", block)}

check(DATA_AND_SECRETS <= gitignore, f"git ignores every data/secret pattern: {sorted(DATA_AND_SECRETS - gitignore)}")
check(DATA_AND_SECRETS <= excluded, f"the push excludes every data/secret pattern: {sorted(DATA_AND_SECRETS - excluded)}")
check(excluded == DATA_AND_SECRETS, f"no unexplained entries in the push's data block: {sorted(excluded ^ DATA_AND_SECRETS)}")
```

## Workbench boundary test — ask rsync

```python
import subprocess, tempfile
from pathlib import Path

WORKBENCH = (".claude/", "CLAUDE.md", "AGENTS.md", "LOG.md", "docs/")   # literal, on purpose
MUST_SHIP = ("core/engine.py", "contract/schema.py", "deploy/RUNBOOK.md")


def shipped_paths() -> set[str]:
    excludes = subprocess.check_output(
        ["bash", "-c", "source deploy/excludes.sh; printf '%s\\n' \"${SHIP_EXCLUDES[@]}\""],
        cwd=ROOT, text=True).split()
    with tempfile.TemporaryDirectory() as dest:
        out = subprocess.check_output(
            ["rsync", "-an", "--out-format=%n", *excludes, f"{ROOT}/", dest], text=True)
    return {line.strip() for line in out.splitlines() if line.strip()}


ship = shipped_paths()
for w in WORKBENCH:
    check(not any(p == w or p.startswith(w) for p in ship), f"workbench never ships: {w}")
for m in MUST_SHIP:
    check(m in ship, f"product file still ships: {m}")
check(not any(p.endswith(".env") for p in ship), "no env file ships")
```

## Gated restore (down-sync) skeleton

```bash
#!/usr/bin/env bash
# deploy/restore_data.sh <tenant> <host> — first-time setup / recovery ONLY.
set -euo pipefail
tenant="$1"; host="$2"
ssh "$host" true || { echo "REFUSED: cannot reach $host — cannot prove nothing is running" >&2; exit 1; }
for unit in app-bot app-web capture@ derive@ backup-up; do
  state="$(ssh "$host" "systemctl list-units --no-legend --state=active,activating '${unit}*'" )" \
    || { echo "REFUSED: could not query $unit on $host" >&2; exit 1; }   # an ssh failure is NOT "stopped"
  if [ -n "$state" ]; then
    echo "REFUSED: $unit is active on $host and owns the live data. Stop it (and its timers) first." >&2
    exit 1
  fi
done
read -r -p "This OVERWRITES $tenant's data on $host from backup. Type the tenant id to continue: " typed
[ "$typed" = "$tenant" ] || { echo "aborted" >&2; exit 1; }
stage="$(mktemp -d)"; trap 'rm -rf "$stage"' EXIT
gcloud storage rsync -r "gs://<bucket-for-$tenant>" "$stage"      # stage on the workstation
rsync -az "$stage/" "$host:/srv/app-data/$tenant/"                # then push once
echo "restored; start the services and verify the data marker"
```

## Up-sync (backup) unit pair

```ini
# deploy/backup-up.service
[Unit]
Description=Back up the tenant data tree (host -> bucket, additive)
OnFailure=app-alert@%n.service
[Service]
Type=oneshot
User=app
EnvironmentFile=/etc/<app>/app.env
ExecStart=/opt/<app>/deploy/backup_up.sh
ProtectSystem=strict
ReadOnlyPaths=/srv/app-data
# gcloud needs a writable config dir under ProtectSystem=strict
Environment=CLOUDSDK_CONFIG=/var/lib/app/gcloud
ReadWritePaths=/var/lib/app/gcloud
```
```ini
# deploy/backup-up.timer
[Timer]
OnCalendar=*-*-* 06,18:00:00 UTC
Persistent=true
RandomizedDelaySec=900
[Install]
WantedBy=timers.target
```
`backup_up.sh` runs `gcloud storage rsync -r <data-root> gs://<bucket>` **without** `--delete`
(prune is an explicit, separate flag). Enable bucket object versioning so a corrupted host
copy cannot silently replace the last good one.
